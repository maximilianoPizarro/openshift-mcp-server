OpenShift MCP Server deployed successfully!

1. Check the deployment:
   oc get pods -n {{ .Values.namespace }} -l app={{ include "openshift-mcp-server.fullname" . }}

2. MCP endpoint (Streamable HTTP):
   http://{{ include "openshift-mcp-server.fullname" . }}.{{ .Values.namespace }}.svc.cluster.local:{{ .Values.service.port }}/mcp

3. Add to OLSConfig:

   apiVersion: ols.openshift.io/v1alpha1
   kind: OLSConfig
   metadata:
     name: cluster
   spec:
     featureGates:
       - MCPServer
     mcpServers:
       - name: {{ .Values.olsConfig.mcpServerName }}
         timeout: {{ .Values.olsConfig.mcpServerTimeout }}
         url: 'http://{{ include "openshift-mcp-server.fullname" . }}.{{ .Values.namespace }}.svc.cluster.local:{{ .Values.service.port }}/mcp'
       {{- if .Values.kubernetesMcp.enabled }}
       - name: kubernetes-mcp-server
         timeout: {{ .Values.olsConfig.mcpServerTimeout }}
         url: 'http://{{ include "openshift-mcp-server.fullname" . }}-k8s-mcp.{{ .Values.namespace }}.svc.cluster.local:{{ .Values.kubernetesMcp.service.port }}/mcp'
       {{- end }}

   Test the proxy:
   LITELLM_HOST=$(oc get route {{ include "openshift-mcp-server.fullname" . }}-litellm -n {{ .Values.namespace }} -o jsonpath='{.spec.host}')

   curl -s https://${LITELLM_HOST}/v1/chat/completions \
     -H "Authorization: Bearer {{ .Values.litellm.masterKey }}" \
     -H "Content-Type: application/json" \
     -d '{"model":"{{ .Values.litellm.model.name }}","messages":[{"role":"user","content":"Hello"}]}'

   Note: The OAuth token expires after ~24h. Refresh with:
   helm upgrade {{ .Release.Name }} ./helm/openshift-mcp-server \
     --set litellm.model.apiKey=$(oc whoami -t)
