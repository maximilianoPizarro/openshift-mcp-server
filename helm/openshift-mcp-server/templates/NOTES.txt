To integrate with OpenShift Lightspeed, add to your OLSConfig:
  spec:
    featureGates:
      - MCPServer
    mcpServers:
      - name: openshift-mcp-server
        timeout: 30
        url: 'http://{{ include "openshift-mcp-server.fullname" . }}.{{ .Values.namespace }}.svc.cluster.local:{{ .Values.service.port }}/mcp'
